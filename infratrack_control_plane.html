Embed InfraTrack in Any Application Like a Package

Install with pip install -e . and connect any service as a controllable tool in seconds:

from infratrack import InfraTrackToolAgent, ToolActionResult

agent = InfraTrackToolAgent(
    app_name="NexusPay Order & Payment Gateway",
    tool_name="NexusPay App SDK Tool Agent",
    endpoint_url="http://127.0.0.1:5055",
    control_plane_url="http://127.0.0.1:8000/api/v1",
)

@agent.action("app_restart_workers", risk="LOW", description="Restart payment settlement workers")
def restart_workers(**kwargs) -> ToolActionResult:
    return ToolActionResult(success=True, action="app_restart_workers", message="Workers restarted")

# Registers application, tool capabilities, and infrastructure resources automatically
agent.connect_to_control_plane(resources=[...])
